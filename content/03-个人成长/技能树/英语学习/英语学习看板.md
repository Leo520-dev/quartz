## 🔥 学习连续性

```dataviewjs
const logs = dv.pages('"03-个人成长/技能树/英语学习/01-学习日志"')
  .sort(l => l.日期, 'desc');

if (logs.length === 0) {
  dv.paragraph("暂无学习日志记录。开始你的第一篇学习日志吧！");
} else {
  // Calculate streak
  let streak = 0;
  const today = dv.date("today");
  let checkDate = today;
  
  for (let log of logs) {
    if (log.日期) {
      const diff = Math.floor((checkDate - log.日期) / (1000 * 60 * 60 * 24));
      if (diff <= 1) {
        streak++;
        checkDate = log.日期;
      } else {
        break;
      }
    }
  }
  
  const lastLog = logs[0];
  const daysSinceLast = lastLog.日期 ? Math.floor((today - lastLog.日期) / (1000 * 60 * 60 * 24)) : 0;
  
  dv.table(
    ["连续性指标", "数值"],
    [
      ["🔥 当前连续打卡", streak + " 天"],
      ["📅 最近学习日期", lastLog.日期 ? lastLog.日期.toFormat("yyyy-MM-dd") : "—"],
      ["⏰ 距今天数", daysSinceLast + " 天前"],
      ["📝 总学习日志", logs.length + " 篇"],
      ["📊 平均频率", logs.length > 0 ? "每 " + Math.round(30 / Math.min(logs.length, 30)) + " 天 1 篇" : "—"]
    ]
  );
  
  if (daysSinceLast === 0) {
    dv.paragraph("✅ **今天已学习！** 保持这个势头！");
  } else if (daysSinceLast === 1) {
    dv.paragraph("⚠️ **昨天学习了，今天还未记录哦！**");
  } else {
    dv.paragraph("🔴 **已经 " + daysSinceLast + " 天没有学习记录了，加油！**");
  }
}
```

const logs = dv.pages('"03-个人成长/技能树/英语学习/01-学习日志"')

## 📚 词书进度汇总（DataviewJS）

```dataviewjs
// Aggregate word book progress from learning logs
const logs = dv.pages('"03-个人成长/技能树/英语学习/01-学习日志"');

// Group by learning content if available
const byContent = {};
logs.forEach(l => {
  const content = l.学习内容 || "未分类";
  if (!byContent[content]) {
    byContent[content] = { count: 0, newWords: 0, reviewWords: 0, time: 0 };
  }
  byContent[content].count++;
  byContent[content].newWords += (l.新词数 || 0);
  byContent[content].reviewWords += (l.复习数 || 0);
  byContent[content].time += (l.学习时长 || 0);
});

if (Object.keys(byContent).length === 0) {
  dv.paragraph("暂无学习日志数据。开始记录学习日志来查看词书进度汇总！");
} else {
  const rows = Object.entries(byContent)
    .sort((a, b) => b[1].newWords - a[1].newWords)
    .map(([content, data]) => [
      content,
      data.count + " 次",
      data.newWords + " 词",
      data.reviewWords + " 词",
      data.time + " min",
      Math.round(data.newWords / Math.max(data.count, 1)) + " 词/次"
    ]);
  
  dv.table(
    ["学习内容", "学习次数", "新词", "复习", "总时长", "效率"],
    rows
  );
}
```
